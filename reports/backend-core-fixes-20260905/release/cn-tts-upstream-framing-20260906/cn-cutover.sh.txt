baseline_matches() {
  local pid
  pid=$(systemctl show castreader -p MainPID --value 2>/dev/null || true)

    [ -n "$pid" ] && [ "$(readlink -f "/proc/$pid/cwd" 2>/dev/null || true)" = "$old" ] &&
    [ "$(systemctl is-active castreader 2>/dev/null || true)" = active ] &&
    [ "$(systemctl is-active nginx 2>/dev/null || true)" = active ] &&
    nginx -t >/dev/null 2>&1 &&
    [ "$(sha256sum /etc/nginx/sites-available/castreader 2>/dev/null | awk '{print $1}')" = a78ce00e3c8d9bf4e489c53d4231aa1e1d378669b7b758192e97aa1920b247de ] &&

    [ "$(sha256sum /etc/systemd/system/castreader.service.d/30-release-id.conf 2>/dev/null | awk '{print $1}')" = 5c7ea9e47f374b28d77caf6d2755e4743b2c2c4f957890832017084db90fca6c ]
}
restore_outer_baseline() {
  local rollback_ok=0 active=0
  ln -sfn "$old" /opt/castreader/current.outer-rollback || rollback_ok=1
  mv -Tf /opt/castreader/current.outer-rollback /opt/castreader/current || rollback_ok=1
  cp -p "$outer_backup/nginx.conf" /etc/nginx/sites-available/castreader || rollback_ok=1
  cp -p "$outer_backup/20-service-route.conf" /etc/systemd/system/castreader.service.d/20-service-route.conf || rollback_ok=1
  cp -p "$outer_backup/27-cn-tts.conf" /etc/systemd/system/castreader.service.d/27-cn-tts.conf || rollback_ok=1
  cp -p "$outer_backup/30-release-id.conf" /etc/systemd/system/castreader.service.d/30-release-id.conf || rollback_ok=1
  systemctl daemon-reload || rollback_ok=1
  nginx -t || rollback_ok=1
  systemctl restart castreader || rollback_ok=1
  systemctl reload nginx || rollback_ok=1
  for _ in $(seq 1 30); do
    if [ "$(systemctl is-active castreader 2>/dev/null || true)" = active ]; then active=1; break; fi
    sleep 1
  done
  [ "$active" = 1 ] || rollback_ok=1
  baseline_matches || rollback_ok=1
  return "$rollback_ok"
}
outer_exit() {
  local status=$?
  local cleanup_backup=1
  trap - EXIT INT TERM HUP
  set +e
  if [ "$outer_mutation_started" = 1 ] && [ "$outer_committed" != 1 ]; then
    if baseline_matches; then
      echo '✓ 内层已恢复旧版；外层未重复重启'
    else
      if ! restore_outer_baseline; then
        status=1
        cleanup_backup=0
        echo "✗ 外层回滚未完整成功；保留 root-only 备份：$outer_backup" >&2
      fi
    fi
  fi
  if [ "$status" = 0 ] && [ "$outer_committed" != 1 ]; then status=1; fi
  if [ "$cleanup_backup" = 1 ]; then
    rm -rf -- "$outer_backup"
  else
    chmod 0700 "$outer_backup" 2>/dev/null || true
  fi
  exit "$status"
}
