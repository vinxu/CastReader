#!/usr/bin/env bash
# Remote, transactional activation for api.castreader.cn and the dedicated
# quickread.castreader.cn authenticated ingress.
#
# Production invocation reads the approved masked authority from stdin. It is
# deliberately absent from argv, curl argv and logs. This script never changes
# any global/.ai service or DNS record.

cn_restore_file() {
  local backup_file=$1
  local destination=$2
  local existed=$3
  if [ "$existed" = "1" ]; then
    install -m 0644 "$backup_file" "$destination"
  else
    rm -f -- "$destination"
  fi
}

cn_wait_service_active() {
  local attempt
  for attempt in $(seq 1 10); do
    if systemctl is-active --quiet castreader; then
      return 0
    fi
    sleep 2
  done
  return 1
}

cn_read_runtime_environment() {
  local process_environment=$1
  CN_RUNTIME_ADMIN_SECRET=''
  CN_RUNTIME_ROUTE=''
  CN_RUNTIME_RELEASE_ID=''
  CN_RUNTIME_VOICE_GIFT_ENABLED=''
  CN_RUNTIME_VOICE_GIFT_SCHEMA_READY=''
  CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY=''
  CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET=''
  CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET=''
  CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY=''
  CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY=''

  while IFS= read -r -d '' entry; do
    case "$entry" in
      MOBILE_ROUTING_ADMIN_SECRET=*)
        CN_RUNTIME_ADMIN_SECRET=${entry#MOBILE_ROUTING_ADMIN_SECRET=}
        ;;
      CASTREADER_SERVICE_ROUTE=*)
        CN_RUNTIME_ROUTE=${entry#CASTREADER_SERVICE_ROUTE=}
        ;;
      CASTREADER_RELEASE_ID=*)
        CN_RUNTIME_RELEASE_ID=${entry#CASTREADER_RELEASE_ID=}
        ;;
      VOICE_GIFT_CN_MINIMAL_ENABLED=*)
        CN_RUNTIME_VOICE_GIFT_ENABLED=${entry#VOICE_GIFT_CN_MINIMAL_ENABLED=}
        ;;
      VOICE_GIFT_CN_MINIMAL_SCHEMA_READY=*)
        CN_RUNTIME_VOICE_GIFT_SCHEMA_READY=${entry#VOICE_GIFT_CN_MINIMAL_SCHEMA_READY=}
        ;;
      VOICE_GIFT_CN_DRAFT_SCHEMA_READY=*)
        CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY=${entry#VOICE_GIFT_CN_DRAFT_SCHEMA_READY=}
        ;;
      VOICE_GIFT_CN_INVITATION_SECRET=*)
        CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET=${entry#VOICE_GIFT_CN_INVITATION_SECRET=}
        ;;
      VOICE_CLONE_WEB_BRIDGE_SECRET=*)
        CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET=${entry#VOICE_CLONE_WEB_BRIDGE_SECRET=}
        ;;
      VOICE_GIFT_DRAFT_CAPABILITY_KEY=*)
        CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY=${entry#VOICE_GIFT_DRAFT_CAPABILITY_KEY=}
        ;;
      VOICE_CLONE_ENCRYPTION_KEY=*)
        CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY=${entry#VOICE_CLONE_ENCRYPTION_KEY=}
        ;;
    esac
  done < "$process_environment"
}

cn_verify_runtime_authority() {
  local expected_fingerprint=$1
  local release=$2
  local proc_root=$3
  local main_pid code attempt draft_capability_key

  CN_RUNTIME_MAIN_PID=''

  main_pid=$(systemctl show --property MainPID --value castreader) || return 1
  if [[ ! "$main_pid" =~ ^[1-9][0-9]*$ ]] ||
    [ ! -r "$proc_root/$main_pid/environ" ]; then
    echo "✗ 运行进程身份不可读，拒绝发布"
    return 1
  fi

  cn_read_runtime_environment "$proc_root/$main_pid/environ"
  if [ "$CN_RUNTIME_ROUTE" != "cn" ] ||
    [ "$CN_RUNTIME_RELEASE_ID" != "$release" ] ||
    [ -z "$CN_RUNTIME_ADMIN_SECRET" ] ||
    [ "$CN_RUNTIME_VOICE_GIFT_ENABLED" != "true" ] ||
    [ "$CN_RUNTIME_VOICE_GIFT_SCHEMA_READY" != "true" ] ||
    { [ "$CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY" != "false" ] &&
      [ "$CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY" != "true" ]; } ||
    [ "${#CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET}" -lt 32 ] ||
    [ "${#CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET}" -lt 32 ] ||
    [ -z "$CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY" ] ||
    [ "$CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET" = "$CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET" ] ||
    [ "$CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET" = "$CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY" ] ||
    [ "$CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET" = "$CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY" ]; then
    echo "✗ 运行进程线路、发布身份或中国区邀请录音配置不匹配，拒绝发布"
    unset CN_RUNTIME_ADMIN_SECRET CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET \
      CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY \
      CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY \
      CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY
    return 1
  fi
  draft_capability_key=$CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY
  if [ -z "$draft_capability_key" ]; then
    draft_capability_key=$CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY
  fi
  if [ "$CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY" = "true" ] &&
    { [ "${#draft_capability_key}" -lt 32 ] ||
      [[ "$draft_capability_key" =~ ^[[:space:]] ]] ||
      [[ "$draft_capability_key" =~ [[:space:]]$ ]]; }; then
    echo "✗ 中国声音邀请草稿 capability 密钥未就绪，拒绝发布"
    unset draft_capability_key CN_RUNTIME_ADMIN_SECRET \
      CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET \
      CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET \
      CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY \
      CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY \
      CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY
    return 1
  fi
  unset draft_capability_key
  if [[ "$CN_RUNTIME_ADMIN_SECRET" == *$'\n'* ]] ||
    [[ "$CN_RUNTIME_ADMIN_SECRET" == *$'\r'* ]] ||
    [[ "$CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET" == *$'\n'* ]] ||
    [[ "$CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET" == *$'\r'* ]] ||
    [[ "$CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET" == *$'\n'* ]] ||
    [[ "$CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET" == *$'\r'* ]] ||
    [[ "$CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY" == *$'\n'* ]] ||
    [[ "$CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY" == *$'\r'* ]] ||
    [[ "$CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY" == *$'\n'* ]] ||
    [[ "$CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY" == *$'\r'* ]]; then
    echo "✗ 运行进程管理或邀请录音凭据格式非法，拒绝发布"
    unset CN_RUNTIME_ADMIN_SECRET CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET \
      CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET \
      CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY \
      CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY
    return 1
  fi

  CN_RUNTIME_HEADER_FILE=$(mktemp "${TMPDIR:-/tmp}/castreader-runtime-gate.XXXXXX")
  chmod 0600 "$CN_RUNTIME_HEADER_FILE"
  {
    printf 'Authorization: Bearer %s\n' "$CN_RUNTIME_ADMIN_SECRET"
    printf 'x-castreader-expected-billing-authority: %s\n' "$expected_fingerprint"
    printf 'x-castreader-release-id: %s\n' "$release"
  } > "$CN_RUNTIME_HEADER_FILE"
  unset CN_RUNTIME_ADMIN_SECRET CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET \
    CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET \
    CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY \
    CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY \
    expected_fingerprint

  for attempt in $(seq 1 10); do
    code=$(curl --silent --output /dev/null --write-out '%{http_code}' \
      --max-time 10 --request POST \
      --header "@$CN_RUNTIME_HEADER_FILE" \
      http://127.0.0.1:3000/api/admin/billing-authority 2>/dev/null || true)
    if [ "$code" = "204" ]; then
      rm -f -- "$CN_RUNTIME_HEADER_FILE"
      CN_RUNTIME_HEADER_FILE=''
      unset CN_RUNTIME_ROUTE CN_RUNTIME_RELEASE_ID \
        CN_RUNTIME_VOICE_GIFT_ENABLED CN_RUNTIME_VOICE_GIFT_SCHEMA_READY
      CN_RUNTIME_MAIN_PID=$main_pid
      return 0
    fi
    # These are deterministic authorization/input/authority failures. Retrying
    # cannot make them safe and would only delay rollback.
    if [ "$code" = "400" ] || [ "$code" = "401" ] || [ "$code" = "409" ]; then
      break
    fi
    sleep 3
  done

  rm -f -- "$CN_RUNTIME_HEADER_FILE"
  CN_RUNTIME_HEADER_FILE=''
  unset CN_RUNTIME_ROUTE CN_RUNTIME_RELEASE_ID \
    CN_RUNTIME_VOICE_GIFT_ENABLED CN_RUNTIME_VOICE_GIFT_SCHEMA_READY
  echo "✗ 运行时 billing authority 校验失败，拒绝发布"
  return 1
}

cn_verify_voice_gift_schema() {
  local proc_root=$1
  local expected_main_pid=$2
  local draft_schema_ready=$3
  local verification_scope=${4:-full}
  local process_environment current_main_pid

  if [[ ! "$expected_main_pid" =~ ^[1-9][0-9]*$ ]] ||
    { [ "$draft_schema_ready" != "false" ] &&
      [ "$draft_schema_ready" != "true" ]; } ||
    { [ "$verification_scope" != "phase0" ] &&
      [ "$verification_scope" != "full" ]; } ||
    [ ! -r "$proc_root/$expected_main_pid/environ" ]; then
    echo "✗ 运行进程数据库身份不可读，拒绝发布"
    return 1
  fi
  process_environment="$proc_root/$expected_main_pid/environ"

  # Read DATABASE_URL inside the helper and decompose it into libpq's PG*
  # environment. The URL is never copied to argv, stdout/stderr or a file, and
  # the audit SQL reaches psql over stdin. psql output is captured and reduced
  # to one fixed marker so connection errors cannot disclose URL components.
  if ! python3 - "$process_environment" "$draft_schema_ready" \
    "$verification_scope" <<'PY'

import subprocess
import sys
import urllib.parse

PHASE0_SCHEMA_MARKER = b'voice-clone-phase0-schema-v1'
PHASE0_SCHEMA_QUERY = r"""
SET statement_timeout = '15s';

  SELECT relation.oid
  FROM pg_catalog.pg_class AS relation
  JOIN pg_catalog.pg_namespace AS namespace
    ON namespace.oid = relation.relnamespace
  WHERE namespace.nspname = 'public'
    AND relation.relname = 'voice_clone_voice'
    AND relation.relkind IN ('r', 'p')

expected_columns(
  column_name,

    ('create_attempt_hash', 'text', false, '', '', ''),
    ('create_lease_expires_at', 'timestamp with time zone', false, '', '', '')

    AND attribute.attname IN (

column_gate AS (

    (SELECT count(*) FROM target) = 1
    AND NOT EXISTS (
      (SELECT * FROM expected_columns EXCEPT SELECT * FROM actual_columns)
      UNION ALL
      (SELECT * FROM actual_columns EXCEPT SELECT * FROM expected_columns)
    ) AS passed
),
expected_checks(constraint_name, normalized_definition) AS (

  FROM target

    AND constraint_value.conname IN (

    )

check_gate AS (
  SELECT
    (SELECT count(*) FROM actual_checks) =
      (SELECT count(*) FROM expected_checks)

        ON actual.constraint_name = expected.constraint_name
      WHERE actual.constraint_name IS NULL

index_gate AS (
  SELECT count(*) = 1
    AND bool_and(
      index_name = 'uidx_voice_clone_voice_create_idempotency'
      AND indisunique
      AND NOT indisprimary
      AND indisvalid
      AND indisready
      AND indislive
      AND access_method = 'btree'
      AND indnkeyatts = 2

def decode(value):
    decoded = urllib.parse.unquote(value, errors='strict')
    if '\x00' in decoded:
        raise ValueError
    return decoded


def read_database_url(path):
    entries = open(path, 'rb').read().split(b'\0')
    prefix = b'DATABASE_URL='
    values = [entry[len(prefix):] for entry in entries if entry.startswith(prefix)]
    if len(values) != 1 or not values[0]:
        raise ValueError
    return values[0].decode('utf-8', errors='strict')


def connection_environment(database_url):
    parsed = urllib.parse.urlsplit(database_url)
    if parsed.scheme not in ('postgres', 'postgresql') or parsed.fragment:
        raise ValueError

    environment = {
        key: value
        for key, value in os.environ.items()
        if key not in ('DATABASE_URL', 'POSTGRES_URL') and not key.startswith('PG')
    }
    if parsed.hostname is not None:
        environment['PGHOST'] = decode(parsed.hostname)
    if parsed.port is not None:
        environment['PGPORT'] = str(parsed.port)
    if parsed.username is not None:
        environment['PGUSER'] = decode(parsed.username)
    if parsed.password is not None:
        environment['PGPASSWORD'] = decode(parsed.password)
    if parsed.path not in ('', '/'):
        environment['PGDATABASE'] = decode(parsed.path[1:])

    seen = set()
    for field in parsed.query.split('&') if parsed.query else ():
        if not field or '=' not in field:
            raise ValueError
        raw_key, raw_value = field.split('=', 1)
        key = decode(raw_key)
        if key in IGNORED_DRIVER_OPTIONS:
            continue
        if key not in QUERY_ENVIRONMENT or key in seen:
            raise ValueError
        seen.add(key)
        environment[QUERY_ENVIRONMENT[key]] = decode(raw_value)

    # Bound the release gate even if the runtime URL requests a longer wait.
    environment['PGCONNECT_TIMEOUT'] = '10'
    environment['PGAPPNAME'] = 'castreader-cn-voice-gift-schema-gate'
    environment['LC_ALL'] = 'C'
    return environment


try:
    child_environment = connection_environment(read_database_url(sys.argv[1]))
    draft_schema_ready = sys.argv[2]
    verification_scope = sys.argv[3]
    if draft_schema_ready not in ('false', 'true'):
        raise ValueError
    if verification_scope not in ('phase0', 'full'):
        raise ValueError

    def verify(query, marker):
        result = subprocess.run(
            [
                'psql',
                '--no-psqlrc',
                '--no-password',
                '--tuples-only',
                '--no-align',
                '--quiet',
                '--set=ON_ERROR_STOP=1',
            ],
            input=query.encode('utf-8'),
            stdout=subprocess.PIPE,
            stderr=subprocess.DEVNULL,
            env=child_environment,
            timeout=30,
            check=False,
        )
        if result.returncode != 0 or result.stdout.strip() != marker:
            raise RuntimeError

    verify(PHASE0_SCHEMA_QUERY, PHASE0_SCHEMA_MARKER)
    if verification_scope == 'full':
        verify(SCHEMA_QUERY, SCHEMA_MARKER)
        if draft_schema_ready == 'true':
            verify(DRAFT_SCHEMA_QUERY, DRAFT_SCHEMA_MARKER)
except Exception:
    raise SystemExit(1)
PY
  then
    if [ "$verification_scope" = "phase0" ]; then
      echo "✗ 中国声音克隆 Phase-0 数据库结构或权限未就绪，拒绝发布"
    else
      echo "✗ 中国声音邀请数据库结构或权限未就绪，拒绝发布"
    fi
    return 1
  fi

  current_main_pid=$(systemctl show --property MainPID --value castreader) || {
    echo "✗ 数据库校验后运行进程身份不可读，拒绝发布"
    return 1
  }
  if [ "$current_main_pid" != "$expected_main_pid" ] ||
    [ ! -r "$proc_root/$current_main_pid/environ" ]; then
    echo "✗ 数据库校验期间运行进程已变化，拒绝发布"
    return 1
  fi
}

cn_verify_staged_voice_gift_routes() {
  local staged_nginx=$1
  if ! python3 - "$staged_nginx" <<'PY'
import re
import sys

raw = open(sys.argv[1], encoding='utf-8').read()


def strip_comments(text):
    output = []
    quote = None
    escaped = False
    comment = False
    for character in text:
        if comment:
            if character == '\n':
                comment = False
                output.append(character)
            continue
        if quote is not None:
            output.append(character)
            if escaped:
                escaped = False
            elif character == '\\':
                escaped = True
            elif character == quote:
                quote = None
            continue
        if character == '#':
            comment = True
        else:
            output.append(character)
            if character in ("'", '"'):
                quote = character
    return ''.join(output)


def matching_block(text, opening_brace):
    depth = 0
    quote = None
    escaped = False
    for index in range(opening_brace, len(text)):
        character = text[index]
        if quote is not None:
            if escaped:
                escaped = False
            elif character == '\\':
                escaped = True
            elif character == quote:
                quote = None
            continue
        if character in ("'", '"'):
            quote = character
        elif character == '{':
            depth += 1
        elif character == '}':
            depth -= 1
            if depth == 0:
                return text[opening_brace + 1:index]
    raise ValueError('unterminated block')


def blocks(text, header_pattern):
    result = []
    for match in re.finditer(header_pattern + r'\s*\{', text, re.MULTILINE):
        opening = text.find('{', match.start(), match.end())
        result.append(matching_block(text, opening))
    return result


def compact(value):
    return ' '.join(value.split())


try:
    effective = strip_comments(raw)
    api_tls = [
        block
        for block in blocks(effective, r'(?<![A-Za-z0-9_-])server')
        if re.search(r'(?<![A-Za-z0-9_-])server_name\s+api\.castreader\.cn\s*;', block)
        and re.search(r'(?<![A-Za-z0-9_-])listen\s+443\s+ssl(?:\s+http2)?\s*;', block)
    ]
    if len(api_tls) != 1:
        raise ValueError('api TLS vhost')
    vhost = api_tls[0]
    operations = (
        'invitation',
        'draft-prepare',
        'draft-preview',
        'fulfill-draft',
    )
    location_headers = [
        compact(match.group(1))
        for match in re.finditer(
            r'(?<![A-Za-z0-9_-])location\s+([^{}]+?)\s*\{', vhost
        )
        if '/api/voice-clone/requests/' in match.group(1)
    ]
    expected_headers = sorted(
        f'= /api/voice-clone/requests/{operation}' for operation in operations
    )
    if sorted(location_headers) != expected_headers:
        raise ValueError('exact operation allowlist')
    for operation in operations:
        path = f'/api/voice-clone/requests/{operation}'
        locations = blocks(
            vhost,
            rf'(?<![A-Za-z0-9_-])location\s*=\s*{re.escape(path)}',
        )
        if len(locations) != 1:
            raise ValueError(path)
        body = locations[0]
        if re.findall(r'(?<![A-Za-z0-9_-])client_max_body_size\s+([^;]+);', body) != ['5m']:
            raise ValueError(f'{path} body ceiling')
        includes = re.findall(r'(?<![A-Za-z0-9_-])include\s+([^;]+);', body)
        if includes != ['/etc/nginx/snippets/nginx-voice-gift-web-proxy.inc']:
            raise ValueError(f'{path} bridge include')
except Exception:
    raise SystemExit(1)
PY
  then
    echo "✗ 中国声音邀请 Web bridge 不是精确四路由，拒绝发布"
    return 1
  fi
}

activate_cn_release() {
  local remote_root=$1
  local release=$2
  local expected_fingerprint=$3
  local nginx_destination=$4
  local systemd_dropin_dir=$5
  local proc_root=$6
  local staged_nginx=$7
  local staged_route=$8
  local route_destination="$systemd_dropin_dir/20-service-route.conf"
  local tts_route_destination="$systemd_dropin_dir/27-cn-tts.conf"
  local release_destination="$systemd_dropin_dir/30-release-id.conf"
  local new_release source_commit pre_activation_main_pid
  local previous_target backup_dir
  local nginx_existed=0 route_existed=0 tts_route_existed=0 release_existed=0 rollback_pending=1
  local CN_RUNTIME_HEADER_FILE=''
  local CN_RUNTIME_ADMIN_SECRET=''
  local CN_RUNTIME_ROUTE=''
  local CN_RUNTIME_RELEASE_ID=''
  local CN_RUNTIME_VOICE_GIFT_ENABLED=''
  local CN_RUNTIME_VOICE_GIFT_SCHEMA_READY=''
  local CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY=''
  local CN_RUNTIME_VOICE_GIFT_INVITATION_SECRET=''
  local CN_RUNTIME_VOICE_GIFT_BRIDGE_SECRET=''
  local CN_RUNTIME_VOICE_GIFT_DRAFT_CAPABILITY_KEY=''
  local CN_RUNTIME_VOICE_CLONE_ENCRYPTION_KEY=''
  local CN_RUNTIME_MAIN_PID=''

  # Canonicalize once so rollback-target containment checks are not confused
  # by platform aliases such as /var -> /private/var.
  remote_root=$(readlink -f "$remote_root" 2>/dev/null || true)
  new_release="$remote_root/$release"

  if [ -z "$remote_root" ] || [[ ! "$release" =~ ^release-[0-9]{14}$ ]] ||
    [[ ! "$expected_fingerprint" =~ ^ba1_[0-9a-f]{20}$ ]] ||
    [ ! -d "$new_release" ] || [ ! -f "$staged_nginx" ] ||
    [ ! -f "$staged_route" ]; then
    echo "✗ 发布输入不完整，拒绝切换"
    return 1
  fi
  source_commit=$(cn_release_manifest_source \
    "$new_release/.castreader-release.json" "$release") || {
    echo "✗ 发布源码清单无效，拒绝切换"
    return 1
  }
  cn_verify_staged_tts_routes "$staged_nginx"
  cn_verify_staged_voice_gift_routes "$staged_nginx"

  previous_target=$(readlink -f "$remote_root/current" 2>/dev/null || true)
  case "$previous_target" in
    "$remote_root"/release-*) ;;
    *)
      echo "✗ 没有可验证的旧版本用于回滚，拒绝切换"
      return 1
      ;;
  esac
  if [ ! -d "$previous_target" ] || [ "$previous_target" = "$new_release" ]; then
    echo "✗ 旧版本回滚目标无效，拒绝切换"
    return 1
  fi

  # Phase 0 is shared by all three CN creation modes. Prove the live database
  # has its exact columns, checks and idempotency index before changing files,
  # symlinks, systemd configuration, nginx configuration or either process.
  pre_activation_main_pid=$(systemctl show \
    --property MainPID --value castreader) || {
    echo "✗ 切换前运行进程身份不可读，拒绝发布"
    return 1
  }
  cn_verify_voice_gift_schema "$proc_root" "$pre_activation_main_pid" \
    false phase0

  backup_dir=$(mktemp -d "${TMPDIR:-/tmp}/castreader-cn-rollback.XXXXXX")
  if [ -f "$nginx_destination" ]; then
    cp -p "$nginx_destination" "$backup_dir/nginx.conf"
    nginx_existed=1
  fi
  if [ -f "$route_destination" ]; then
    cp -p "$route_destination" "$backup_dir/route.conf"
    route_existed=1
  fi
  if [ -f "$tts_route_destination" ]; then
    cp -p "$tts_route_destination" "$backup_dir/tts-route.conf"
    tts_route_existed=1
  fi
  if [ -f "$release_destination" ]; then
    cp -p "$release_destination" "$backup_dir/release.conf"
    release_existed=1
  fi

  cn_rollback_current_activation() {
    local rollback_ok=0
    set +e
    trap - EXIT ERR INT TERM HUP
    [ -n "$CN_RUNTIME_HEADER_FILE" ] && rm -f -- "$CN_RUNTIME_HEADER_FILE"
    ln -sfn "$previous_target" "$remote_root/current.rollback" &&
      mv -Tf "$remote_root/current.rollback" "$remote_root/current" ||
      rollback_ok=1
    cn_restore_file "$backup_dir/nginx.conf" "$nginx_destination" \
      "$nginx_existed" || rollback_ok=1
    cn_restore_file "$backup_dir/route.conf" "$route_destination" \
      "$route_existed" || rollback_ok=1
    cn_restore_file "$backup_dir/tts-route.conf" "$tts_route_destination" \
      "$tts_route_existed" || rollback_ok=1
    cn_restore_file "$backup_dir/release.conf" "$release_destination" \
      "$release_existed" || rollback_ok=1
    systemctl daemon-reload || rollback_ok=1
    nginx -t || rollback_ok=1
    systemctl restart castreader || rollback_ok=1
    systemctl reload nginx || rollback_ok=1
    cn_wait_service_active || rollback_ok=1
    rm -rf -- "$backup_dir"
    if [ "$rollback_ok" = "0" ]; then
      echo "✓ 已完整回滚到旧版中国区服务"
    else
      echo "✗ 自动回滚未完整成功，需要立即人工恢复中国区服务"
    fi
    return "$rollback_ok"
  }

  cn_activation_failure() {
    local failure_status=$1
    trap - EXIT ERR INT TERM HUP
    if [ "$failure_status" = "0" ]; then failure_status=1; fi
    if [ "$rollback_pending" = "1" ]; then
      cn_rollback_current_activation || true
    fi
    exit "$failure_status"
  }
  # EXIT covers explicit exits and shell termination paths that do not raise ERR.
  # ERR/INT/TERM/HUP keep the original immediate rollback behavior.
  trap 'cn_activation_failure $?' EXIT
  trap 'cn_activation_failure $?' ERR
  trap 'cn_activation_failure 130' INT
  trap 'cn_activation_failure 143' TERM
  trap 'cn_activation_failure 129' HUP

  chown -R castreader:castreader "$new_release"
  install -m 0644 "$staged_nginx" "$nginx_destination"
  install -d -m 0755 "$systemd_dropin_dir"
  install -m 0644 "$staged_route" "$route_destination"
  # The previous CN hotfix also installed a later 27-cn-tts.conf. Keep this
  # managed override in the same transaction; otherwise it silently supersedes
  # the newly approved route in 20-service-route.conf after daemon-reload.
  {
    printf '[Service]\n'
    sed -n '/^Environment=TTS_UPSTREAM_URL=/p' "$staged_route"
  } > "$tts_route_destination"
  chmod 0644 "$tts_route_destination"
  printf '[Service]\nEnvironment=CASTREADER_RELEASE_ID=%s\n' "$release" \
    > "$release_destination"
  chmod 0644 "$release_destination"
  nginx -t

  ln -sfn "$new_release" "$remote_root/current.new"
  mv -Tf "$remote_root/current.new" "$remote_root/current"
  systemctl daemon-reload
  systemctl restart castreader
  systemctl reload nginx
  cn_wait_service_active

  # Both gates are tied to the restarted Node process. The first proves the
  # enabled release identity; the second rechecks Phase 0 and audits the live
  # invitation tables before the rollback commit point.
  cn_verify_runtime_authority "$expected_fingerprint" "$release" "$proc_root"
  cn_verify_voice_gift_schema "$proc_root" "$CN_RUNTIME_MAIN_PID" \
    "$CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY"
  cn_verify_public_health
  cn_verify_quickread_public_health
  cn_verify_voice_clone_contract "$source_commit" \
    "$CN_RUNTIME_VOICE_GIFT_DRAFT_SCHEMA_READY"
  cn_verify_tts_public_health

  rollback_pending=0
  trap - EXIT ERR INT TERM HUP
  rm -rf -- "$backup_dir"
  # This continuity release deliberately retains every old immutable release.
  # Cleanup is a separate reviewed operation after the observation window.
  echo "✓ 已保留全部旧发布目录与精确回滚目标"
  echo "✓ 中国区发布已通过运行时账本、声音克隆 Phase-0/邀请数据库、声音克隆合同与入口健康校验"
}

main() {
  local expected_fingerprint
  set +x
  if [ "$#" -ne 2 ]; then
    echo "✗ activate-cn-release 需要 remote root 与 release id"
    return 2
  fi
  IFS= read -r expected_fingerprint || true
  if [[ ! "$expected_fingerprint" =~ ^ba1_[0-9a-f]{20}$ ]]; then
    echo "✗ 缺少有效的审批账本标识，拒绝切换"
    return 2
  fi
  activate_cn_release \
    "$1" "$2" "$expected_fingerprint" \
    /etc/nginx/sites-available/castreader \
    /etc/systemd/system/castreader.service.d \
    /proc \
    /tmp/castreader-nginx.conf \
    /tmp/castreader-cn-route.conf
  unset expected_fingerprint
}

if [[ "${BASH_SOURCE[0]}" == "$0" ]]; then
  set -Eeuo pipefail
  main "$@"
fi
